set(PC_SANITIZER "none" CACHE STRING "none, address (plus UB), undefined, or thread")
set_property(CACHE PC_SANITIZER PROPERTY STRINGS none address undefined thread)
if(NOT PC_SANITIZER MATCHES "^(none|address|undefined|thread)$")
    message(FATAL_ERROR "Invalid PC_SANITIZER")
endif()

add_library(pc_a1_options INTERFACE)
target_compile_features(pc_a1_options INTERFACE cxx_std_20)
target_include_directories(pc_a1_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(pc_a1_options INTERFACE Threads::Threads)
if(MSVC)
    target_compile_options(pc_a1_options INTERFACE /W4 /permissive-)
    if(NOT PC_SANITIZER STREQUAL "none")
        message(FATAL_ERROR "Sanitizer presets require GCC or Clang")
    endif()
else()
    target_compile_options(pc_a1_options INTERFACE -Wall -Wextra -Wpedantic)
    if(NOT PC_SANITIZER STREQUAL "none")
        set(sanitizer_flags "-fsanitize=${PC_SANITIZER}")
        if(PC_SANITIZER STREQUAL "address")
            set(sanitizer_flags "-fsanitize=address,undefined")
        endif()
        target_compile_options(pc_a1_options INTERFACE ${sanitizer_flags} -fno-omit-frame-pointer -fno-sanitize-recover=all)
        target_link_options(pc_a1_options INTERFACE ${sanitizer_flags})
    endif()
endif()

# The runner checks this fingerprint against the working tree before measuring.
file(GLOB_RECURSE kernel_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND kernel_sources "${PROJECT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT kernel_sources)
set(source_catalog "")
foreach(source IN LISTS kernel_sources)
    file(SHA256 "${source}" digest)
    file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}" "${source}")
    string(APPEND source_catalog "${relative}:${digest}\n")
endforeach()
string(SHA256 source_digest "${source_catalog}")
target_compile_definitions(pc_a1_options INTERFACE
    PC_SOURCE_SHA256="${source_digest}"
    PC_BUILD_TYPE="$<CONFIG>"
    PC_SANITIZER_NAME="${PC_SANITIZER}")

add_library(pc_workload STATIC src/workload.cpp)
target_link_libraries(pc_workload PUBLIC pc_a1_options)
set_property(TARGET pc_workload PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
foreach(suite IN ITEMS locks barriers counters)
    add_executable(bench_${suite} benchmarks/bench_${suite}.cpp)
    target_link_libraries(bench_${suite} PRIVATE pc_workload)
    set_property(TARGET bench_${suite} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
endforeach()
add_executable(pc_environment src/environment.cpp)
target_link_libraries(pc_environment PRIVATE pc_a1_options)
