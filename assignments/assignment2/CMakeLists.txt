set(PC_SANITIZER "none" CACHE STRING "none, address, undefined, or thread")
if(NOT PC_SANITIZER MATCHES "^(none|address|undefined|thread)$")
    message(FATAL_ERROR "Invalid PC_SANITIZER")
endif()
add_library(pc_a2_options INTERFACE)
target_include_directories(pc_a2_options INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(pc_a2_options INTERFACE Threads::Threads)
if(MSVC)
    target_compile_options(pc_a2_options INTERFACE /W4 /permissive-)
    if(NOT PC_SANITIZER STREQUAL "none")
        message(FATAL_ERROR "Sanitizers require GCC/Clang")
    endif()
else()
    target_compile_options(pc_a2_options INTERFACE -Wall -Wextra -Wpedantic)
    if(NOT PC_SANITIZER STREQUAL "none")
        set(flags "-fsanitize=${PC_SANITIZER}")
        if(PC_SANITIZER STREQUAL "address")
            set(flags "-fsanitize=address,undefined")
        endif()
        target_compile_options(pc_a2_options INTERFACE ${flags} -fno-omit-frame-pointer -fno-sanitize-recover=all)
        target_link_options(pc_a2_options INTERFACE ${flags})
    endif()
endif()
file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(APPEND sources "${PROJECT_SOURCE_DIR}/CMakeLists.txt" "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT sources)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${sources})
set(catalog "")
foreach(source IN LISTS sources)
    file(SHA256 "${source}" digest)
    file(RELATIVE_PATH relative "${PROJECT_SOURCE_DIR}" "${source}")
    string(APPEND catalog "${relative}:${digest}\n")
endforeach()
string(SHA256 source_digest "${catalog}")
target_compile_definitions(pc_a2_options INTERFACE PC_A2_SOURCE_SHA256="${source_digest}" PC_A2_BUILD_TYPE="$<CONFIG>" PC_A2_SANITIZER="${PC_SANITIZER}")
add_library(pc_a2 STATIC src/graph.cpp src/search.cpp)
target_link_libraries(pc_a2 PUBLIC pc_a2_options)
add_executable(a2_route src/route_main.cpp)
target_link_libraries(a2_route PRIVATE pc_a2)
add_executable(a2_bench_set src/set_main.cpp)
target_link_libraries(a2_bench_set PRIVATE pc_a2_options)
if(BUILD_TESTING)
    foreach(suite IN ITEMS lists search)
        add_executable(a2_test_${suite} tests/test_${suite}.cpp)
        target_link_libraries(a2_test_${suite} PRIVATE pc_a2)
        add_test(NAME a2_${suite} COMMAND a2_test_${suite})
        set_tests_properties(a2_${suite} PROPERTIES TIMEOUT 180)
    endforeach()
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME a2_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py" "$<TARGET_FILE_DIR:a2_route>" "${CMAKE_CURRENT_SOURCE_DIR}/data/tradeoff.road")
        add_test(NAME a2_tools COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py")
        set_tests_properties(a2_cli a2_tools PROPERTIES TIMEOUT 180)
    endif()
endif()
